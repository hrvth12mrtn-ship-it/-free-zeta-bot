import logging
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes
import google.generativeai as genai

logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO
)

TELEGRAM_TOKEN = "8154369506:AAGxZmRnuRA5l8f29g04pXgN3i18h4-8M5c"
GEMINI_API_KEY = "AQ.Ab8RN6JXLMAlAyraANI2yO5Bv0Wz9e11M7v8U3Kj"

genai.configure(api_key=GEMINI_API_KEY)

SYSTEM_INSTRUCTION = """
Te egy segítőkész AI asszisztens vagy Telegramon. Válaszolj a felhasználóknak kedvesen és érthetően.
"""

model = genai.GenerativeModel(
    model_name="gemini-1.5-flash",
    system_instruction=SYSTEM_INSTRUCTION
)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("Szia! Én vagyok a te AI botod. Kérdezz tőlem bármit!")

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_message = update.message.text
    try:
        response = model.generate_content(user_message)
        await update.message.reply_text(response.text)
    except Exception as e:
        await update.message.reply_text("Hiba történt a válaszadás közben.")

def main():
    application = Application.builder().token(TELEGRAM_TOKEN).build()
    application.add_handler(CommandHandler("start", start))
    application.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
    application.run_polling()

if __name__ == '__main__':
    main()
